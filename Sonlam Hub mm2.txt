local function Create(class, parent, props)
   local obj = Instance.new(class, parent)
   for k, v in pairs(props) do obj[k] = v end
   return obj
end

local function AddCorner(parent, radius)

local function BindTouchHold(btn, setVar)
   btn.InputBegan:Connect(function(input)
       if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
           setVar(true)
       end
   end)
   btn.InputEnded:Connect(function(input)
       if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
           setVar(false)
       end
   end)
end

   btn.MouseButton1Click:Connect(function()

local function GetRole(p)
   if not p.Character then return Color3.fromRGB(0, 255, 0), "Innocent" end
   
   local function hasKnife(container)
       if not container then return false end
       for _, item in ipairs(container:GetChildren()) do
           if item:IsA("Tool") then
               local lname = string.lower(item.Name)
               if lname:find("knife") or lname:find("dagger") or lname:find("blade") or item:FindFirstChild("KnifeServer") or item:FindFirstChild("Stab") then
                   return true
               end
           end
       end
       return false
   end

   local function hasGun(container)
       if not container then return false end
       for _, item in ipairs(container:GetChildren()) do
           if item:IsA("Tool") then
               local lname = string.lower(item.Name)
               if lname:find("gun") or lname:find("revolver") or lname:find("pistol") or item:FindFirstChild("ShootGun") then
                   return true
               end
           end
       end
       return false
   end

   if hasKnife(p.Character) or hasKnife(p.Backpack) then
       return Color3.fromRGB(255, 0, 0), "Murderer"
   end
   if hasGun(p.Character) or hasGun(p.Backpack) then
       return Color3.fromRGB(0, 0, 255), "Sheriff"
   end
   return Color3.fromRGB(0, 255, 0), "Innocent"
end

local function IsInLobby(target)
   local pos
   if typeof(target) == "Instance" then
       if target:IsA("Model") and target:FindFirstChild("HumanoidRootPart") then
           pos = target.HumanoidRootPart.Position
       elseif target:IsA("BasePart") then
           pos = target.Position
       end
   elseif typeof(target) == "Vector3" then
       pos = target
   end
   if not pos then return true end

   local lobby = Workspace:FindFirstChild("Lobby")
   if lobby then
       for _, part in ipairs(lobby:GetDescendants()) do
           if part:IsA("BasePart") then
               if (pos - part.Position).Magnitude < 60 then
                   return true
               end
           end
       end
   end

   if pos.Y < -20 or pos.Y > 350 then
       return true
   end

   return false
end

               local timeToHit = math.clamp(distance / 280, 0, 0.12)
               local predictedPos = targetHrp.Position + (flatVel * timeToHit) + Vector3.new(0, 0.5, 0)
               Camera.CFrame = Camera.CFrame:Lerp(CFrame.lookAt(Camera.CFrame.Position, predictedPos), 0.75)
           end
       end)
   end

   if _G.Noclip and LP.Character then
       for _, part in pairs(LP.Character:GetDescendants()) do
           if part:IsA("BasePart") and part.CanCollide then
               part.CanCollide = false
           end
       end
   end
   
   if _G.Flying and LP.Character then
       local hrp = LP.Character:FindFirstChild("HumanoidRootPart")
       local hum = LP.Character:FindFirstChildOfClass("Humanoid")
       local flyGyro = hrp and hrp:FindFirstChild("FlyGyro")
       local flyVel = hrp and hrp:FindFirstChild("FlyVel")
       
       if hrp and hum and flyGyro and flyVel then
           local camCFrame = Camera.CFrame
           local moveVec = Vector3.zero

           if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVec = moveVec + camCFrame.LookVector end
           if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec - camCFrame.LookVector end
           if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVec = moveVec - camCFrame.RightVector end
           if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVec = moveVec + camCFrame.RightVector end

           if moveVec.Magnitude == 0 and hum.MoveDirection.Magnitude > 0 then
               local camLook = camCFrame.LookVector
               local camRight = camCFrame.RightVector
               local flatCamLook = Vector3.new(camLook.X, 0, camLook.Z).Unit
               local flatCamRight = Vector3.new(camRight.X, 0, camRight.Z).Unit
               local forwardAmount = hum.MoveDirection:Dot(flatCamLook)
               local rightAmount = hum.MoveDirection:Dot(flatCamRight)
               moveVec = (camLook * forwardAmount) + (camRight * rightAmount)
           end

           if isFlyingUp then moveVec = moveVec + Vector3.new(0, 1, 0) end
           if isFlyingDown then moveVec = moveVec - Vector3.new(0, 1, 0) end

           if moveVec.Magnitude > 0 then
               flyVel.Velocity = moveVec.Unit * _G.FlySpeed
           else
               flyVel.Velocity = Vector3.zero
           end
           
           flyGyro.CFrame = camCFrame
       end
   end
end)
